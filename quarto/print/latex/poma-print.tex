% Pandoc switches to main matter immediately after the contents. The print
% edition has publisher front matter after the contents, so defer that first
% automatic switch; each volume's opening publisher note invokes the real
% \mainmatter immediately before the authored journey.
\let\pomarealmainmatter\mainmatter
\newif\ifpomamainmatterrequested
\pomamainmatterrequestedfalse
\renewcommand{\mainmatter}{%
  \ifpomamainmatterrequested
    \pomarealmainmatter
  \else
    \global\pomamainmatterrequestedtrue
  \fi}
